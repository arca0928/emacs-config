  (setup-define :package
    (lambda (package) nil)
    :shorthand 'cadr
    :repeatable 't)
#+end_src

#+begin_src elisp
  (setup-define :opt
    (lambda (&rest pairs) `(setopt ,@pairs))
    :after-loaded t)
#+end_src

#+begin_src elisp
  (setup-define :global
    (lambda (key command)
      `(keymap-global-set ,key ,command))
    :repeatable t
    :ensure '(nil func))

  	corfu-preselect 'prompt
