#+begin_src elisp
  (setup (:package corfu)
    (:require corfu)
    (:opt corfu-cycle t
  	corfu-auto t
  	corfu-auto-delay 0.1
  	corfu-auto-prefix 2
  	corfu-preselect 'prompt
  	corfu-on-exact-match nil)
    (:with-map corfu-map
      (:bind
       "C-p" corfu-previous
       "C-n" corfu-next
       "C-y" corfu-insert
       "C-e" corfu-quit
       "M-SPC" corfu-insert-separator
       "C-b" corfu-popupinfo-scroll-down
       "C-f" corfu-popupinfo-scroll-up)
      (:unbind "RET"))
    (global-corfu-mode 1))

  (setup corfu-popupinfo
    (:require corfu)
    (corfu-popupinfo-mode 1))
